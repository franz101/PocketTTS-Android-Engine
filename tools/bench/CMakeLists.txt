# Standalone ARM64 CLI build of PocketTTS.cpp for on-device benchmarking over adb.
# Uses the same ONNX Runtime binary/headers as the app (scripts/prepare_android_native_deps.sh).
cmake_minimum_required(VERSION 3.22.1)
project(pockettts_bench LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ORT_LIB "${ORT_LIB_OVERRIDE}")
if(NOT ORT_LIB)
  set(ORT_LIB "${ROOT}/app/src/main/jniLibs/${ANDROID_ABI}/libonnxruntime.so")
endif()
set(ORT_HEADERS "${ORT_HEADERS_OVERRIDE}")
if(NOT ORT_HEADERS)
  set(ORT_HEADERS "${ROOT}/app/src/main/third_party/onnxruntime/include")
endif()
include(FetchContent)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_PROTOBUF_PROVIDER "internal" CACHE STRING "" FORCE)
set(SPM_ABSL_PROVIDER "internal" CACHE STRING "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sentencepiece GIT_REPOSITORY https://github.com/google/sentencepiece.git GIT_TAG v0.2.1 GIT_SHALLOW TRUE)
FetchContent_Declare(dr_libs GIT_REPOSITORY https://github.com/mackron/dr_libs.git GIT_TAG 50bb723e6a459dbb781e26cefee4fd9ca6714d6a)
FetchContent_MakeAvailable(sentencepiece dr_libs)
target_compile_options(sentencepiece-static PRIVATE -w)
foreach(t spm_decode spm_encode spm_export_vocab spm_normalize spm_train)
  if(TARGET ${t})
    target_link_libraries(${t} log)
  endif()
endforeach()
add_executable(pockettts_cli "${ROOT}/vendor/PocketTTS.cpp/pocket_tts.cpp")
target_compile_options(pockettts_cli PRIVATE -O3 -w)
target_include_directories(pockettts_cli PRIVATE "${ORT_HEADERS}" "${ORT_HEADERS}/onnxruntime/core/session"
  "${sentencepiece_SOURCE_DIR}/src" "${dr_libs_SOURCE_DIR}")
target_link_libraries(pockettts_cli PRIVATE "${ORT_LIB}" sentencepiece-static log)
